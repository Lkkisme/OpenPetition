% --- Bidirectional Exhibit Citation & Back-Reference Engine ---
\makeatletter
\newcommand{\citeex}[2]{%
  \leavevmode
  \begingroup
  % Record current page in the exhibit's back-reference list
  \@ifundefined{exbackref@#1}{%
    \expandafter\xdef\csname exbackref@#1\endcsname{\thepage}%
  }{%
    \expandafter\xdef\csname exbackref@#1\endcsname{\csname exbackref@#1\endcsname, \thepage}%
  }%
  \hyperref[exhibit:#1]{\textbf{[Exhibit~#1\if\relax\detokenize{#2}\relax\else, #2\fi]}}%
  \endgroup
}

\newcommand{\getexbackrefs}[1]{%
  \@ifundefined{exbackref@#1}{Referenced in Exhibit Index}{\csname exbackref@#1\endcsname}%
}
\makeatother

% --- Scanner-Proof Exhibit Cover Sheet Macro ---
\newcommand{\exhibitcoversheet}[4]{%
  \clearpage
  \phantomsection
  \label{exhibit:#1}%
  \addcontentsline{toc}{subsection}{Exhibit #1 — #2}%
  \vspace*{0.25in}
  \begin{center}
    {\LARGE\bfseries EXHIBIT #1}\\[0.6em]
    {\Large\bfseries #2}\\[1.2em]
  \end{center}
  \begin{tcolorbox}[colback=gray!5,colframe=black!75,boxrule=0.8pt,arc=2pt]
    \renewcommand{\arraystretch}{1.35}
    \begin{tabular}{@{}p{0.26\textwidth}p{0.68\textwidth}@{}}
      \textbf{Petitioner / Beneficiary:} & \PetitionerName \\
      \textbf{Classification:} & \PetitionClassification \\
      \textbf{Statutory / Regulatory Hook:} & #3 \\
      \textbf{Cited on Brief Page(s):} & \getexbackrefs{#1} \\
      \textbf{Officer Highlighting Guide:} & #4 \\
    \end{tabular}
  \end{tcolorbox}
  \vspace{0.2in}
}
